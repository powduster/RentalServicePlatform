use myapp;

create table user (
UserID varchar(50) primary key, #회원 아이디:중복불가
UserPW varchar(255), #해쉬로 변환해서 저장
UserName varchar(50), #닉네임:중복가능
Email varchar(50), #이메일 문자 및 본인 확인 용도
JoiningDay datetime, #가입날짜
Balance float #지갑
)

create table Item(
ItemName varchar(50), #물건 이름
UserID varchar(50),
PostingDay datetime, #물건 게시일
ClosingDay datetime, #물건 마감일
presence bool, #물건의 유무
price float, #가격
Discribtion varchar(255), #설명칸
foreign key(UserID) references user(UserID)
);
